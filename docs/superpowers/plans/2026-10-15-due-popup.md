# Due date pop-up Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans (the user chose native execution) to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Set an item's due date from a calendar icon (or its due text) that opens a pop-up, and make edit mode change only the text.

**Architecture:** Two new routes (`GET`/`PUT /items/{id}/due`) and a `due-picker` template give the pop-up, a `<form>` that htmx submits. Two new service functions (`SetDue`, `EditText`) replace `Edit`/`EditIn`. The label list's open/close/place code in `app.js` becomes shared by both pop-ups.

**Tech Stack:** Go 1.27 (`net/http`, `html/template`), htmx 2, SQLite store, plain JS and CSS.

**Spec:** `docs/superpowers/specs/2026-10-15-due-popup-design.md` (in the worktree). Design: canvas "Todo app UI proposal", page "10 · Due date", board "NEW: calendar icon and due date pop-up", option A.

## Review Focus

1. Clicking the calendar icon or due text of an open pop-up closes it, and does not close and reopen it (focusout race).
2. A click on a non-focusable part of the due pop-up (title, hint, gap) does not close it.
3. A half-typed date, or a time like `25:00`, does not save as "no due date" (native form validation blocks the submit).
4. After a 422 the new pop-up keeps focus in the date field, and the label list still gets its cursor at the end of the search.
5. Opening the due pop-up while the label list is open closes the label list, and the other way around.

```go
// EditText changes only the text of an item. The due date, its notified
// state, and the labels stay.
func (s *Service) EditText(ctx context.Context, userID, id int64, text string) (Item, error) {
	text, err := cleanText(text)
	if err != nil {
		return Item{}, err
	}
	it, err := s.store.Get(ctx, userID, id)
	if err != nil {
		return Item{}, err
	}
	if it.Text == text {
		return it, nil
	}
	return s.store.UpdateItem(ctx, userID, id,
		Change{Text: text, DueAt: it.DueAt, DueAllDay: it.DueAllDay, KeepLabels: true}, s.now())
}

// SetDue changes only the due date of an item, read in loc (nil: the
// service's zone). An empty date and time removes the due date. A new due
// date resets the notified state, so the item can notify again.
func (s *Service) SetDue(ctx context.Context, userID, id int64, date, clock string, loc *time.Location) (Item, error) {
	dueAt, allDay, err := parseDue(date, clock, s.zone(loc))
	if err != nil {
		return Item{}, err
	}
	it, err := s.store.Get(ctx, userID, id)
	if err != nil {
		return Item{}, err
	}
	if sameDue(it.DueAt, it.DueAllDay, dueAt, allDay) {
		return it, nil
	}
	return s.store.UpdateItem(ctx, userID, id,
		Change{Text: it.Text, DueAt: dueAt, DueAllDay: allDay, ResetNotified: true, KeepLabels: true}, s.now())
}
```

```go
package web

import (
	"errors"
	"net/http"
	"strconv"

	"todo/internal/todo"
)

// duePicker shows the due date pop-up that the calendar icon opens.
func (s *server) duePicker(w http.ResponseWriter, r *http.Request) {
	item, ok := s.findItem(w, r)
	if !ok {
		return
	}
	date, clock := dueInputs(item, s.zoneFor(r))
	s.render(w, r, http.StatusOK, part{"due-picker", duePickerView{Item: item, DueDate: date, DueTime: clock}})
}

// setItemDue sets the due date of an item from the pop-up, or removes it
// when both fields are empty. The answer is the row, or the whole list if
// the item moves; on a bad value it is the pop-up with a message.
func (s *server) setItemDue(w http.ResponseWriter, r *http.Request) {
	id, ok := parseID(r)
	if !ok {
		s.oobOnly(w, r, http.StatusNotFound)
		return
	}
	date, clock := r.FormValue("due_date"), r.FormValue("due_time")
	// Read the time in the zone the pop-up was shown in, so an unchanged
	// time stays the same moment even if the browser's zone changed since.
	loc, ok := s.parseZone(r.FormValue("due_tz"))
	if !ok {
		loc = s.zoneFor(r)
	}
	old := s.oldItem(r, id)
	item, err := s.svc.SetDue(r.Context(), userID(r), id, date, clock, loc)
	var dueErr *todo.DueError
	switch {
	case errors.As(err, &dueErr):
		cur, getErr := s.svc.Get(r.Context(), userID(r), id)
		if errors.Is(getErr, todo.ErrNotFound) {
			s.oobOnly(w, r, http.StatusNotFound)
			return
		}
		if getErr != nil {
			s.serverError(w, r, getErr)
			return
		}
		w.Header().Set("HX-Retarget", "#due-picker-"+strconv.FormatInt(id, 10))
		w.Header().Set("HX-Reswap", "outerHTML")
		s.render(w, r, http.StatusUnprocessableEntity,
			part{"due-picker", duePickerView{Item: cur, DueDate: date, DueTime: clock, Error: dueErr.Msg}})
	case errors.Is(err, todo.ErrNotFound):
		s.oobOnly(w, r, http.StatusNotFound)
	case err != nil:
		s.serverError(w, r, err)
	default:
		if placeChanged(old, item, s.nowFor(r)) {
			s.renderList(w, r, currentUser(r).HideDone)
			return
		}
		// The permission bar depends on the due dates, so it comes along.
		lv, err := s.listView(r, currentUser(r).HideDone)
		if err != nil {
			s.serverError(w, r, err)
			return
		}
		lv.OOB = true
		s.render(w, r, http.StatusOK, part{"item", item}, part{"oob", lv})
	}
}
```

```js
let picker = null; // the open pop-up

function isDuePicker() {

}

// focusPicker puts the cursor in the first field of the pop-up: the search
// of the label list, or the date of the due date pop-up.
function focusPicker() {
  const input = picker.querySelector('input:not([type="hidden"])');
  if (input) {
    input.focus();
  }
  return input;
}

// popupButton is the + or calendar icon that a press on target belongs to.

function popupButton(target) {
  if (!target.closest) {
    return null;
  }
  const due = target.closest("button.due");
  return due ? due.parentElement.querySelector(".due-add") : target.closest(".label-add, .due-add");
```
